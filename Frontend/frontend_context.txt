interface CategoryPieChartProps {
  breakdown: Record<string, number>;
  currency?: string;
  size?: number;
}

interface Slice {
  category: string;
  value: number;
  percentage: number;
  color: string;
}

function polarToCartesian(cx: number, cy: number, r: number, angleDeg: number) {
  const angleRad = ((angleDeg - 90) * Math.PI) / 180;
  return {
    x: cx + r * Math.cos(angleRad),
    y: cy + r * Math.sin(angleRad),
  };
}

function describeArc(
  cx: number,
  cy: number,
  r: number,
  startAngle: number,
  endAngle: number,
) {
  const start = polarToCartesian(cx, cy, r, endAngle);
  const end = polarToCartesian(cx, cy, r, startAngle);
  const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1';
  return [
    `M ${cx} ${cy}`,
    `L ${start.x} ${start.y}`,
    `A ${r} ${r} 0 ${largeArcFlag} 0 ${end.x} ${end.y}`,
    'Z',
  ].join(' ');
}

export const CategoryPieChart: React.FC<CategoryPieChartProps> = ({
  breakdown,
  currency = 'USD',
  size = 220,
}) => {
  const { colors, spacing } = useTheme();
  const entries = Object.entries(breakdown).filter(([, v]) => v > 0);
  const total = entries.reduce((sum, [, v]) => sum + v, 0);

  if (!entries.length || total <= 0) {
    return (
      <View
        style={{
          height: size,
          alignItems: 'center',
          justifyContent: 'center',
        }}
      >
        <AppText variant="body" color="muted">
          No spending data yet
        </AppText>
      </View>
    );
  }

  let cursor = 0;
  const slices: Slice[] = entries
    .sort((a, b) => b[1] - a[1])
    .map(([category, value], index) => {
      const percentage = (value / total) * 100;
      const slice: Slice = {
        category,
        value,
        percentage,
        color: colors.chartPalette[index % colors.chartPalette.length],
      };
      return slice;
    });

  const radius = size / 2;
  const strokeInset = 2;

  return (
    <View>
      <View style={{ alignItems: 'center' }}>
        <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
          {slices.map(slice => {
            const angleSpan = (slice.value / total) * 360;
            const startAngle = cursor;
            const endAngle = cursor + angleSpan;
            cursor = endAngle;
            return (
              <Path
                key={slice.category}
                d={describeArc(
                  radius,
                  radius,
                  radius - strokeInset,
                  startAngle,
                  endAngle,
                )}
                fill={slice.color}
                stroke={colors.surface}
                strokeWidth={2}
              />
            );
          })}
          <Circle cx={radius} cy={radius} r={radius * 0.55} fill={colors.surface} />
        </Svg>
        <View style={{ position: 'absolute', alignItems: 'center' }}>
          <AppText variant="caption" color="muted">
            Total Spent
          </AppText>
          <AppText variant="numericMedium">
            {formatCurrency(total, currency)}
          </AppText>
        </View>
      </View>

      <View style={{ marginTop: spacing.md }}>
        {slices.map(slice => {

  register: (name: string, email: string, password: string) => Promise<void>;

export const AuthProvider: React.FC<PropsWithChildren> = ({ children }) => {

  const [token, setToken] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  const logout = useCallback(async () => {
    setUser(null);
    setToken(null);
    await Promise.all([
      removeItem(StorageKeys.AUTH_TOKEN),
      removeItem(StorageKeys.AUTH_USER),
    ]);
  }, []);

  // Bootstrap session from storage on app start.
  useEffect(() => {
    (async () => {
      const [storedToken, storedUser] = await Promise.all([
        getItem<string>(StorageKeys.AUTH_TOKEN),
        getItem<User>(StorageKeys.AUTH_USER),
      ]);
      if (storedToken && storedUser) {
        setToken(storedToken);
        setUser(storedUser);
      }
      setIsLoading(false);
    })();
  }, []);

  // Global 401 handler â€” any API call that comes back unauthorized logs the user out.
  useEffect(() => {
    const unsubscribe = appEvents.on(APP_EVENTS.UNAUTHORIZED, () => {
      logout();
    });
    return unsubscribe;
  }, [logout]);

  const login = useCallback(async (email: string, password: string) => {
    try {
      const result = await authService.login({ email, password });
      setToken(result.token);
      setUser(result.user);
      await Promise.all([
        setItem(StorageKeys.AUTH_TOKEN, result.token),
        setItem(StorageKeys.AUTH_USER, result.user),
      ]);
    } catch (error) {
      throw new Error(parseApiError(error).message);
    }
  }, []);

  const register = useCallback(
    async (name: string, email: string, password: string) => {
      try {
        await authService.register({ name, email, password });
        // Backend register doesn't return a token â€” log in right after.
        await login(email, password);
      } catch (error) {
        throw new Error(parseApiError(error).message);
      }
    },
    [login],
  );

    () => ({
      user,
      token,
      isLoading,
      isAuthenticated: !!token,
      login,
      register,
      logout,
    }),
    [user, token, isLoading, logout, login, register],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;

export function useExpenses(autoFetch = true): UseExpensesResult {
  const [expenses, setExpenses] = useState<Expense[]>([]);
  const [loading, setLoading] = useState(autoFetch);
  const [refreshing, setRefreshing] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchExpenses = useCallback(async (isRefresh = false) => {
    if (isRefresh) setRefreshing(true);
    else setLoading(true);
    setError(null);
    try {
      const result = await expenseService.getExpenses();
      // Most recent first.
      result.sort(
        (a, b) =>
          new Date(b.expenseDate).getTime() - new Date(a.expenseDate).getTime(),
      );
      setExpenses(result);
    } catch (err) {
      setError(parseApiError(err).message);
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, []);

  useEffect(() => {
    if (autoFetch) {
      fetchExpenses();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const createExpense = async (payload: CreateExpenseRequest) => {
    const created = await expenseService.createExpense(payload);
    setExpenses(prev =>
      [created, ...prev].sort(
        (a, b) =>
          new Date(b.expenseDate).getTime() - new Date(a.expenseDate).getTime(),
      ),
    );
    return created;
  };

  const updateExpense = async (id: string, payload: UpdateExpenseRequest) => {
    const updated = await expenseService.updateExpense(id, payload);
    setExpenses(prev => prev.map(e => (e.id === id ? updated : e)));
    return updated;
  };

  const deleteExpense = async (id: string) => {
    await expenseService.deleteExpense(id);
    setExpenses(prev => prev.filter(e => e.id !== id));
  };

  return {
    expenses,
    loading,
    refreshing,
    error,
    refresh: () => fetchExpenses(true),
    createExpense,
    updateExpense,
    deleteExpense,
  };
}

  updateProfile: (
    monthlyIncome: number,
    savingsGoal: number,
  ) => Promise<UserProfile>;
}

export function useProfile(autoFetch = true): UseProfileResult {
  const [profile, setProfile] = useState<UserProfile | null>(null);
  const [loading, setLoading] = useState(autoFetch);
  const [error, setError] = useState<string | null>(null);
  const [hasNoProfile, setHasNoProfile] = useState(false);

  const refresh = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const result = await profileService.getProfile();
      setProfile(result);
      setHasNoProfile(false);
    } catch (err) {
      if (axios.isAxiosError(err) && err.response?.status === 404) {
        setProfile(null);
        setHasNoProfile(true);
      } else {
        setError(parseApiError(err).message);
      }
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    if (autoFetch) {
      refresh();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const createProfile = async (monthlyIncome: number, savingsGoal: number) => {
    const result = await profileService.createProfile({
      monthlyIncome,
      savingsGoal,
    });
    setProfile(result);
    setHasNoProfile(false);
    return result;
  };

  const updateProfile = async (monthlyIncome: number, savingsGoal: number) => {
    const result = await profileService.updateProfile({
      monthlyIncome,
      savingsGoal,
    });
    setProfile(result);
    return result;
  };

  return {
    profile,
    loading,
    error,
    hasNoProfile,
    refresh,
    createProfile,
    updateProfile,
  };
}

export async function getItem<T>(key: string): Promise<T | null> {
  try {
    const raw = await AsyncStorage.getItem(key);
    if (raw === null) return null;
    try {
      return JSON.parse(raw) as T;
    } catch {
      // Value was stored as a plain string (e.g. the raw token).
      return raw as unknown as T;
    }
  } catch (error) {
    console.warn(`[storage] Failed to read "${key}"`, error);
    return null;
  }
}

export async function setItem<T>(key: string, value: T): Promise<void> {
  try {
    const raw = typeof value === 'string' ? value : JSON.stringify(value);
    await AsyncStorage.setItem(key, raw);
  } catch (error) {
    console.warn(`[storage] Failed to write "${key}"`, error);
  }
}

interface ThemeContextValue {
  mode: ThemeMode;
  isDark: boolean;
  colors: ThemeColors;
  typography: typeof typography;
  spacing: typeof spacing;
  radius: typeof radius;
  shadow: typeof shadow;
  setMode: (mode: ThemeMode) => void;
}

const ThemeContext = createContext<ThemeContextValue | undefined>(undefined);

export const ThemeProvider: React.FC<PropsWithChildren> = ({ children }) => {
  const [mode, setModeState] = useState<ThemeMode>('system');
  const [systemScheme, setSystemScheme] = useState<ColorSchemeName>(
    () => Appearance.getColorScheme() ?? 'light',
  );

  useEffect(() => {
    (async () => {
      const saved = await getItem<ThemeMode>(StorageKeys.THEME_MODE);
      if (saved) {
        setModeState(saved);
      }
    })();
  }, []);

  useEffect(() => {
    const sub = Appearance.addChangeListener(({ colorScheme }) => {
      setSystemScheme(colorScheme);
    });
    return () => sub.remove();
  }, []);

  const setMode = (next: ThemeMode) => {
    setModeState(next);
    setItem(StorageKeys.THEME_MODE, next);
  };

  const isDark =
    mode === 'dark' || (mode === 'system' && systemScheme === 'dark');

  const value = useMemo<ThemeContextValue>(
    () => ({
      mode,
      isDark,
      colors: isDark ? darkColors : lightColors,
      typography,
      spacing,
      radius,
      shadow,
      setMode,
    }),
    [mode, isDark],
  );

  return (
    <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>
  );
};

export const useTheme = (): ThemeContextValue => {
  const ctx = useContext(ThemeContext);
  if (!ctx) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return ctx;

type Listener = (...args: any[]) => void;

class EventEmitter {
  private listeners: Record<string, Listener[]> = {};

  on(event: string, listener: Listener): () => void {
    if (!this.listeners[event]) this.listeners[event] = [];
    this.listeners[event].push(listener);
    return () => this.off(event, listener);
  }

  off(event: string, listener: Listener): void {
    this.listeners[event] = (this.listeners[event] || []).filter(
      l => l !== listener,
    );
  }

  emit(event: string, ...args: any[]): void {
    (this.listeners[event] || []).forEach(l => l(...args));
  }
}

/**
 * Looks for the most likely "total" amount on a receipt. Receipts print many
 * numbers (subtotal, tax, line items) â€” we bias toward lines containing a
 * total-like keyword, and fall back to the largest currency-shaped number
 * found anywhere in the text.
 */
export function parseAmountFromReceipt(text: string): number | null {
  const lines = text.split('\n');
  const totalKeywords = /\b(total|amount due|grand total|balance due|amount)\b/i;
  const currencyNumber = /(?:[$â‚¬Â£â‚¹]\s?)?(\d{1,3}(?:[,.\s]\d{3})*(?:[.,]\d{2})?)/g;

  let bestFromKeywordLine: number | null = null;
  for (const line of lines) {
    if (totalKeywords.test(line)) {
      const matches = [...line.matchAll(currencyNumber)];
      if (matches.length) {
        const last = matches[matches.length - 1][1];
        const value = normalizeNumber(last);
        if (value !== null) {
          bestFromKeywordLine = value;
        }
      }
    }
  }
  if (bestFromKeywordLine !== null) return bestFromKeywordLine;

  // Fallback: largest plausible currency amount in the whole receipt.
  const allMatches = [...text.matchAll(currencyNumber)]
    .map(m => normalizeNumber(m[1]))
    .filter((n): n is number => n !== null && n > 0 && n < 100000);

  if (!allMatches.length) return null;
  return Math.max(...allMatches);
}

function normalizeNumber(raw: string): number | null {
  // Strip thousands separators, keep last dot/comma as decimal point.
  const cleaned = raw.replace(/\s/g, '');
  const lastComma = cleaned.lastIndexOf(',');
  const lastDot = cleaned.lastIndexOf('.');
  let normalized = cleaned;
  if (lastComma > lastDot) {
    normalized = cleaned.replace(/\./g, '').replace(',', '.');
  } else {
    normalized = cleaned.replace(/,/g, '');
  }
  const value = parseFloat(normalized);
  return Number.isFinite(value) ? value : null;
}

const DATE_PATTERNS: RegExp[] = [
  /\b(\d{4})[-/](\d{1,2})[-/](\d{1,2})\b/, // yyyy-mm-dd
  /\b(\d{1,2})[-/](\d{1,2})[-/](\d{2,4})\b/, // dd-mm-yyyy or mm-dd-yyyy
  /\b(\d{1,2})\s+(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*\s+(\d{2,4})\b/i,
];

export function parseDateFromReceipt(text: string): Date | null {
  for (const pattern of DATE_PATTERNS) {
    const match = text.match(pattern);
    if (!match) continue;

    if (pattern === DATE_PATTERNS[0]) {
      const [, y, m, d] = match;
      const date = new Date(Number(y), Number(m) - 1, Number(d));
      if (isValidDate(date)) return date;
    } else if (pattern === DATE_PATTERNS[1]) {
      const [, a, b, y] = match;
      const year = y.length === 2 ? 2000 + Number(y) : Number(y);
      // Assume day/month order (common outside the US); if day > 12 this is
      // unambiguous, otherwise it's a best guess.
      const day = Number(a) > 12 ? Number(a) : Number(a);
      const month = Number(a) > 12 ? Number(b) : Number(a);
      const date = new Date(year, (Number(a) > 12 ? Number(b) : month) - 1, day);
      if (isValidDate(date)) return date;
    } else {
      const [, d, mon, y] = match;
      const year = y.length === 2 ? 2000 + Number(y) : Number(y);
      const monthIndex = MONTHS.findIndex(m =>
        mon.toLowerCase().startsWith(m),
      );
      if (monthIndex >= 0) {
        const date = new Date(year, monthIndex, Number(d));
        if (isValidDate(date)) return date;
      }
    }
  }
  return null;
}

const MONTHS = [
  'jan', 'feb', 'mar', 'apr', 'may', 'jun',
  'jul', 'aug', 'sep', 'oct', 'nov', 'dec',
];

function isValidDate(date: Date): boolean {
  return !Number.isNaN(date.getTime()) && date <= new Date();
}

export function suggestCategoryFromReceipt(text: string): ExpenseCategory {
  const lower = text.toLowerCase();
  let bestCategory: ExpenseCategory = 'OTHER';
  let bestScore = 0;

  (Object.keys(CATEGORY_KEYWORDS) as ExpenseCategory[]).forEach(category => {
    const keywords = CATEGORY_KEYWORDS[category];
    const score = keywords.reduce(
      (count, keyword) => (lower.includes(keyword) ? count + 1 : count),
      0,
    );
    if (score > bestScore) {
      bestScore = score;
      bestCategory = category;
    }
  });

  return bestCategory;
}
